            this.date = date;
            this.open = open;
            this.high = high;
            this.low = low;
            this.close = close;
            this.uptrend = uptrend;

        }

    public static class DataFrame {
        public List<OHLCRow> rows = new ArrayList<>();

            rows.add(row);
        }

        public OHLCRow get(int index) {
            return rows.get(index);
        }

        public int size() {
            return rows.size();
        }

            DataFrame df = new DataFrame();
            for (OHLCRow row : rows) {
                df.rows.add(new OHLCRow(row.date, row.open, row.high, row.low, row.close, row.uptrend));
            }
            return df;
        }

        public DataFrame tail(int n) {
            DataFrame result = new DataFrame();
            int start = Math.max(0, rows.size() - n);
            for (int i = start; i < rows.size(); i++) {
                result.rows.add(rows.get(i));
            }
            return result;
        }

        public double minOf(String column) {
            double min = Double.MAX_VALUE;

    /**
     * Line Break chart indicator.

    public static class LineBreak extends Instrument {
        protected DataFrame cdf;
        public int lineNumber = 3;

        public LineBreak(DataFrame df) {
            super(df);
        }

        public boolean uptrendReversal(double close) {
            if (cdf.rows.size() < lineNumber) {
                return false;
            }
            DataFrame lastN = cdf.tail(lineNumber);
            return close < lastN.minOf("low");
        }

        public boolean downtrendReversal(double close) {
            if (cdf.rows.size() < lineNumber) {
                return false;
            }
            DataFrame lastN = cdf.tail(lineNumber);
            return close > lastN.maxOf("high");
        }

        public DataFrame getOhlcData() {
            cdf = new DataFrame();
            int initRows = Math.min(lineNumber, df.rows.size());
            for (int idx = 0; idx < initRows; idx++) {
                OHLCRow row = df.rows.get(idx);
                OHLCRow newRow = new OHLCRow(row.date, row.open, row.high, row.low, row.close, true);
                newRow.index = idx;
                cdf.rows.add(newRow);
            }

            if (df.rows.size() <= lineNumber) {
                return cdf;
            }

            int lastIdx = cdf.rows.size() - 1;
            boolean uptrend = cdf.rows.get(lastIdx).close >= cdf.rows.get(0).open;

            for (int idx = lineNumber; idx < df.rows.size(); idx++) {
                OHLCRow row = df.rows.get(idx);
                double closePrice = row.close;
                lastIdx = cdf.rows.size() - 1;
                double lastClose = cdf.rows.get(lastIdx).close;
                double lastOpen = cdf.rows.get(lastIdx).open;

                if (uptrend) {
                    if (closePrice > lastClose) {
                        OHLCRow newRow = new OHLCRow(row.date, lastClose, closePrice, lastClose, closePrice, true);
                        newRow.index = idx;
                        cdf.rows.add(newRow);
                    } else if (uptrendReversal(closePrice)) {
                        uptrend = false;
                        OHLCRow newRow = new OHLCRow(row.date, lastClose, lastClose, closePrice, closePrice, false);
                        newRow.index = idx;
                        cdf.rows.add(newRow);
                    }
                } else {
                    if (closePrice < lastClose) {
                        OHLCRow newRow = new OHLCRow(row.date, lastClose, lastClose, closePrice, closePrice, false);
                        newRow.index = idx;
                        cdf.rows.add(newRow);
                    } else if (downtrendReversal(closePrice)) {
                        uptrend = true;
                        OHLCRow newRow = new OHLCRow(row.date, lastClose, closePrice, lastClose, closePrice, true);
                        newRow.index = idx;
                        cdf.rows.add(newRow);
                    }
                }
            }
            return cdf;
        }
    }

    /**
     * Point and Figure chart indicator.
     */
    public static class PnF extends Instrument {
        protected DataFrame cdf;
        public double boxSize = 2;
        public int reversalSize = 3;

        public PnF(DataFrame df) {
            super(df);
        }

        public Integer getState(boolean uptrendP1, int bricks) {
            if (bricks == 0) {
                return null;
            }
            if (uptrendP1) {
                return bricks > 0 ? UPTREND_CONTINUAL : UPTREND_REVERSAL;
            } else {
                return bricks < 0 ? DOWNTREND_CONTINUAL : DOWNTREND_REVERSAL;
            }
        }

        public int roundIt(double x, int base) {
            return (int) (base * Math.round(x / base));
        }

        public DataFrame getOhlcData() {
            cdf = new DataFrame();
            double firstValue = df.rows.get(0).close;
            double firstOpen = roundIt(firstValue, (int) boxSize) - boxSize;
            double firstClose = firstOpen + boxSize;

            cdf.rows.add(new OHLCRow(df.rows.get(0).date, firstOpen, firstClose, firstOpen, firstClose, true));

            double currentClose = firstClose;
            double currentOpen = firstOpen;
            boolean uptrend = true;

            for (int idx = 1; idx < df.rows.size(); idx++) {
                OHLCRow row = df.rows.get(idx);
                double price = row.close;
                String date = row.date;

                if (uptrend) {
                    while (price >= currentClose + boxSize) {
                        double newOpen = currentClose;
                        double newClose = newOpen + boxSize;
                        cdf.rows.add(new OHLCRow(date, newOpen, newClose, newOpen, newClose, true));
                        currentOpen = newOpen;
                        currentClose = newClose;
                    }
                    double reversalThreshold = currentOpen - reversalSize * boxSize;
                    if (price < reversalThreshold) {

                        double newClose = newOpen - boxSize;
                        cdf.rows.add(new OHLCRow(date, newOpen, newOpen, newClose, newClose, false));

                            currentClose = newClose;
                        }
                    }
                } else {
                    while (price <= currentClose - boxSize) {

                        double newClose = newOpen - boxSize;
                        cdf.rows.add(new OHLCRow(date, newOpen, newOpen, newClose, newClose, false));
